I did it!!!! Here is a Pokemon themed art thing. Press the number of the brush you want to use before clicking your mouse.
Gotta catch 'em all!
Brushes: 1- Bulbasaur aka the bestest boi
2-Charmander
3- Pikachu
4-Squirtle
5- action star
6- Pokeball to catch your guy
7- green dot to draw grass
8- brown dot to draw the ground
9- a pink flower to add detail to your scene
0- picks a random color, that you can use for the background


p5.js - JavaScript 
var img;
var initials ='ads'; // your initials
var choice = '1'; // starting choice, so it is not empty
var screenbg = 250; // off white background
var lastscreenshot=61; // last screenshot never taken

function preload() {
// preload() runs once, it may make you wait
//  img = loadImage('cat.jpg');  // cat.jpg needs to be next to this .js file
// you can link to an image on your github account
  bulbaimg = loadImage('https://bookittie.github.io/images/bulba.png');
  charmanderimg = loadImage('https://bookittie.github.io/images/charmander.png');
  pikapikaimg = loadImage('https://bookittie.github.io/images/pikapika.png');
  squamimg = loadImage('https://bookittie.github.io/images/squam.png');
  pokeimg = loadImage('https://bookittie.github.io/images/poke_ball.png');
}

function setup() {
createCanvas(800, 800);  // canvas size
background(screenbg);   // use our background screen color

}

function draw() {
  if (keyIsPressed) {
    choice = key; // set choice to the key that was pressed
    clear_print(); // check to see if it is clear screen or save image
  }
  if (mouseIsPressed){
    newkeyChoice(choice);  // if the mouse is pressed call newkeyChoice
  }
}

function newkeyChoice(toolChoice) { //toolchoice is the key that was pressed
  // the key mapping if statements that you can change to do anything you want.
  // just make sure each key option has the a stroke or fill and then what type of 
  // graphic function

 if (toolChoice == '1' ) {  // first brush- bulbasaur
   
    tint(255, 80); // color change (none) and transparency
    image(bulbaimg, mouseX-225, mouseY-225, 450, 450);
    
  } else if (toolChoice == '2') { // second tool- charmander

    tint(255, 80); // color change (none) and transparency
    image(charmanderimg, mouseX-225, mouseY-225, 450, 450);
  } else if (toolChoice == '3') { // third tool- pikachu

   tint(255, 80); // color change (none) and transparency
    image(pikapikaimg, mouseX-225, mouseY-225, 450, 450);
  } else if (toolChoice == '4') { // fourth tool- squirtle

    tint(255, 80); // color change (none) and transparency
    image(squamimg, mouseX-225, mouseY-225, 450, 450);
  } else if (key == '5') { // this tool calls a function - draws action star
    drawAction(mouseX, mouseY);
    
 // make testbox do something!

  } else if (toolChoice == '6') { //draws pokeball

    tint(255, 80); // color change (none) and transparency
    image(pokeimg, mouseX-225, mouseY-225, 450, 450);
  } else if (toolChoice == '7') { //draws green dot for making grass

  } else if (toolChoice == '8') { //draws brown dots for ground making

    fill(131, 101, 57);

  } else if (toolChoice == '9') { // Tool 9: pink and yellow flower Stamp (Click to place flowers)

  } else if (toolChoice == '0') {
    stroke(0, 0);

function drawAction(x, y) {
  push();
  translate(x, y);
  noStroke();
  // star
  fill(90, 110, 135); 
  beginShape();
  vertex(53, 31) //top of star;
  vertex(58, 54) //clockwise down;
  vertex(75, 39);
  vertex(64, 58);
  vertex(86, 59);
  vertex(66, 67);
  vertex(82, 81);
  vertex(61, 75);
  vertex(65, 96);
  vertex(53, 78);
  vertex(42, 96);
  vertex(45, 75);
  vertex(24, 81);
  vertex(40, 67);
  vertex(20, 59);
  vertex(42, 58);
  vertex(32, 39);
  vertex(49, 53);
  endShape(CLOSE);
  pop();
}

function testbox(r, g, b) {
// this is a test function that will show you how you can put your own functions into the sketch
  x = mouseX;
  y = mouseY;
  fill(r, g, b);
  rect(x-50, y-50, 100, 100);

}

function clear_print() {
// this will do one of two things, x clears the screen by resetting the background
// p calls the routine saveme, which saves a copy of the screen
  if (key == 'x' || key == 'X') {
    background(screenbg); // set the screen back to the background color
  } else if (key == 'p' || key == 'P') {
     saveme();  // call saveme which saves an image of the screen
  }
}

function saveme(){
    //this will save the name as the intials, date, time and a millis counting number.
    // it will always be larger in value then the last one.
  filename=initials+day() + hour() + minute() +second();
  if (second()!=lastscreenshot) { // don't take a screenshot if you just took one
    saveCanvas(filename, 'jpg');
    key="";
  }
  lastscreenshot=second(); // set this to the current second so no more than one per second
  
}
